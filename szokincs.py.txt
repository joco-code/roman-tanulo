def hangot_lejatszik(szoveg, nyelv):
    """Létrehoz egy átmeneti hangfájlt és lejátsza."""
    try:
        tts = gTTS(text=szoveg, lang=nyelv, slow=False)
        fajlnev = "temp_hang.mp3"
        tts.save(fajlnev)
        
        # Operációs rendszer szerinti lejátszás
        if platform.system() == "Darwin": # macOS
            subprocess.run(["afplay", fajlnev])
        elif platform.system() == "Windows":
            os.system(f"start {fajlnev}")
        else: # Linux
            subprocess.run(["mpg123", fajlnev])
    except Exception as e:
        print(f"Hiba a hang lejátszásakor: {e}")
